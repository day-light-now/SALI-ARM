cmake_minimum_required(VERSION 3.22)
project(slaihand_hardware)
find_package(ament_cmake REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(OpenArmCAN REQUIRED)
find_package(slaihand_transport REQUIRED)
find_package(std_srvs REQUIRED)
find_package(rcl_interfaces REQUIRED)
add_library(${PROJECT_NAME} SHARED src/openarm_joint_fake.cpp src/canusb_readonly.cpp src/motion_supervisor.cpp src/motion_audit.cpp src/gravity_ff.cpp src/canusb_system.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(${PROJECT_NAME} PRIVATE include)
target_link_libraries(${PROJECT_NAME} OpenArmCAN::openarm_can slaihand_transport::slaihand_transport)
ament_target_dependencies(${PROJECT_NAME} hardware_interface pluginlib rclcpp rclcpp_lifecycle std_msgs std_srvs rcl_interfaces)
pluginlib_export_plugin_description_file(hardware_interface slaihand_hardware.xml)
install(TARGETS ${PROJECT_NAME} DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
