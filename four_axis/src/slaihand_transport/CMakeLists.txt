cmake_minimum_required(VERSION 3.22)
project(slaihand_transport)
find_package(ament_cmake REQUIRED)
add_library(${PROJECT_NAME} SHARED src/canusb_readonly.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(${PROJECT_NAME} PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME} LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
if(BUILD_TESTING)
  enable_testing()
  add_executable(canusb_codec_test test/canusb_codec_test.cpp)
  target_link_libraries(canusb_codec_test ${PROJECT_NAME})
  add_test(NAME canusb_readonly_protocol COMMAND canusb_codec_test)
endif()
ament_package()
